
核心思想：利用递归，不停把数组分成两个，然后合并时顺便排序，排序是用三个指针分别把两个数组更小的放进新数组中

- ①使用时直接调用mergeSort(nums,left,right)，这是选定[left,right]部分排序  

- ②过程分析：
- 大概就是先递后归，递时把数组拆成两部分，归时合并数组顺便排序。
- 先一直往深处递，不停的给数组分成两部分，直到终于l<=r，这时候返回。只是return，啥也不做，对原本的数组还没啥影响，就是终止了继续往深处的行为。  
- 例如[6]和[3]遇到return返回，只是退回到[6,3]这一层。当时在往深处递的时候，就是进行到了"mergeSort(nums, left, mid);mergeSort(nums, mid + 1, right);"这一层将其分为[6]和[3]，现在两个都返回来了，相当于这两个没动过，然后就顺着代码顺序往下进行merge(nums, left, mid, right);，给排序好，变成[3,6]。  
- 合并之后，这一个函数执行到头，退出，在上一层函数中继续往后执行，即"mergeSort(nums, left, mid);mergeSort(nums, mid + 1, right);"拆分数组的部分，接着再和同一层刚返回的其他数组进行merge的排序  

```cpp
// 把已排序好的两个有序区间合并成大的有序数组
void merge(vector<int> &nums, int left, int mid, int right)
{
    // 创建临时数组存储合并结果
    vector<int> temp(right - left + 1);
    int i = left; // 左半部分[left,mid]，左部分的索引
    int j = mid + 1; // 右半部分[mid+1,right]，右部分的索引
    int k = 0; // 临时数组的索引

    // 比较左右两边，谁小谁先放入temp
    while (i <= mid && j <= right)
    {
        if (nums[i] <= nums[j])
        {
            // 左边的部分放入，并且索引后移一个
            temp[k] = nums[i];
            k++;
            i++;
        }
        else
        {
            // 右边的部分放入，并且索引后移一个
            temp[k] = nums[j];
            k++;
            j++;
        }
    }

    // 由于上面的条件是左右部分的索引都不超过边界，那么结束后就说明至少有一个部分的索引超过边界，即已经放完所有元素到临时数组  
    // 若是其中一个数组超出边界，那么另一个数组剩下的都是比这个数组更大的，直接放入  
    // 若是两个数组都超出边界，那么不用管  
  
    // 处理左半部分剩余元素  
    while (i <= mid)
    {
        temp[k] = nums[i];
        k++;
        i++;
    }

    // 处理右半部分剩余元素
    while (j <= right)
    {
        temp[k] = nums[j];
        k++;
        j++;
    }

    // 把临时数组内容拷回原数组
    for (int m = 0; m < temp.size(); m++)
    {
        nums[left + m] = temp[m];
    }
}

void mergeSort(vector<int> &nums, int left, int right)
{
    if (left >= right)
        return; // 只剩一个元素，有序

    int mid = left + (right - left) / 2; // 找中点

    // 分治：分别排序左右两半
    mergeSort(nums, left, mid);
    mergeSort(nums, mid + 1, right);

    // 合并：将两个有序子数组合并
    merge(nums, left, mid, right);
}
```


实际运用：用归并排序来求逆序对：
一个数组[6,3,8,2,9,1]的逆序对有[6,3],[6,2],[6,1],[3,2],[3,1],[8,2],[8,1],[2,1],[9,1]

尝试讲解思路：
调用时使用countInversions(nums,left,right)，这是要排序数组nums的[left,right]部分，该函数的实现过程是：在浅一层使用countInversions把数组拆分为两个，再使用往深层递继续countInversions拆分，直到往深处递时遇到left>=right(即数组只有一个元素)时返回深一层数组的逆序对数之后，统计这左右两个部分内部的逆序对数，这个深一层的左右两个部分内部逆序对是通过long long leftCount = countInversions(nums, left, mid);这条代码内的countInversions中往深处递的mergeCount求出来的，在左右两个部分都求出了内部的逆序对后，因为左右之间可能存在逆序对，所以需要用mergeCount来求，左右分别排序好了再求左右的逆序对，或者不排序直接求左右之间的逆序对，其实都是可以的，因为排序了再求比较方便所以就这样了。  

在mergeCount函数内，对于它外面的countInversions函数中的数组对半分并各自排序有再利用，三个指针ijk分别指向左部分、右部分和新数组，i指向的<=j指向的，放入新数组；若>，放入新数组，顺便统计逆序对个数，因为i指向的部分后面都是更大的，所以只要j小于它，就同时小于了它后面的部分，即(mid-i+1)个，下面再进行剩余部分处理，单组内剩下的不可能有逆序对，排序完之后更新数组，返回这两个已排序数组之间的逆序对总数。  

最后回到最浅层。总的来看，大致过程就是，不断拆分，直到不能再拆，就开始返回，返回过程中顺便统计逆序对个数(深一层的左右部分内部逆序对总数+它们之间的逆序对总数)，直到回到最浅层返回结果

```cpp
long long mergeCount(vector<int> &nums, int left, int mid, int right)
{
    vector<int> temp(right - left + 1);
    int i = left, j = mid + 1, k = 0;
    long long count = 0; // 逆序对计数

    while (i <= mid && j <= right)
    {
        if (nums[i] <= nums[j])
        {
            temp[k++] = nums[i++];
        }
        else
        {
            // 关键点：nums[j] < nums[i]
            // 那么nums[i...mid]都比nums[j]大，都能和nums[j]组成逆序对
            count += (mid - i + 1);
            temp[k++] = nums[j++];
        }
    }

    // 下面这部分都是单组内剩下的，不会再有逆序对了
    while (i <= mid)
        temp[k++] = nums[i++];
    while (j <= right)
        temp[k++] = nums[j++];

    for (int m = 0; m < temp.size(); m++)
    {
        nums[left + m] = temp[m];
    }

    return count;
}

long long countInversions(vector<int> &nums, int left, int right)
{
    if (left >= right)
        return 0;

    int mid = left + (right - left) / 2;
    long long leftCount = countInversions(nums, left, mid);       // 深一层拆成两个的数组在返回时合并且排序，顺便统计逆序对
    long long rightCount = countInversions(nums, mid + 1, right); // 同上
    long long mergeResult = mergeCount(nums, left, mid, right);   // 合并排序并统计逆序对

    return leftCount + rightCount + mergeResult;
}
```
