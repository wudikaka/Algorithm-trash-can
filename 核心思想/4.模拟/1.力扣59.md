
给你一个正整数 n ，生成一个包含 1 到 n2 所有元素，且元素按顺时针顺序螺旋排列的 n x n 正方形矩阵 matrix 。  

***四指针/模拟***
***方向数组***

我的解法：  
- 可以直接绕圈给每个位置都赋值一遍，可以视为最外层绕一圈，然后往里缩再绕一圈，一直如此。为了保证赋值不重复或越界，需要用类似于四指针的思想，startx、starty、end和end四个限制住了四个角，从startx、starty开始，在n-1结束，即 < n，执行完最后一个之后还会有++，所以要--抵消，从左往右和从上往下执行完之后还停留在最后一个刚刚赋完值的位置，所以要i++和j--来调整  

```cpp
class Solution
{
public:
    vector<vector<int>> generateMatrix(int n)
    {
        // 定义一个矩阵
        vector<vector<int>> res(n, vector<int>(n, 0));
        // 定义每次循环一个圈的起始位置
        // 往左和往上 回归起点右下角
        int startx = 0;
        int starty = 0;
        // 总共走几圈
        int loop = n / 2;
        // 矩阵中心位置
        int mid = n / 2; //(n/2,n/2)
        // 当前该赋的值
        int count = 1;
        // 需要控制每条边遍历的长度，往右和往下，每次循环后收缩一次
        // 如第一轮移动n-1个，第二轮移动n-2个
        int end = n;

        int i, j;
        while (loop--)
        {
            i = startx; // 表示行
            j = starty; // 表示列 ----> 不同于数学里的xy轴!!!

            // 下面四个for就是模拟转了一圈
            // 从左往右
            for (j; j < end; j++)
            {
                res[i][j] = count;
                count++;
            }
            j--;
            i++;
            // 从上往下
            for (i; i < end; i++)
            {
                res[i][j] = count;
                count++;
            }
            i--;
            j--;
            // 从右往左
            for (j; j > starty; j--)
            {
                res[i][j] = count;
                count++;
            }
            // 从下往上
            for (i; i > startx; i--)
            {
                res[i][j] = count;
                count++;
            }

            // 更新下一轮的起始位置
            startx++;
            starty++;

            // 更新下一轮遍历长度
            end--;
        }
        // 如果n为奇数，最中心那个走圈赋值走不到，要单独赋值
        if (n % 2 != 0)
        {
            res[mid][mid] = count;
        }

        return res;
    }
};
```

- 官方解法二：  
先创建两个表示方向的数组，相同下标的情况下dx和dy配合，达成向右、向下、向左、向上的目的，即0:右, 1:下, 2:左, 3:上。  
先从(0,0)开始，每次填充完，都先尝试让x、y移动，再判断是否越界/重复，若发现不符合条件，则修改方向，由于方向只可能是右下左上 右下左上循环，所以可以确定修改方向为什么，所以可以直接让方向标准+1，修改为预设的方向，这里为了保证在0~3循环，用了dir=(dir+1)%4，非常牛逼。为了安全，用while而不是if，因为不确定是否可以一次成功，但理论上转向必定成功。  
如果没有进入while转向，可以直接让nx、ny赋值给x、y；如果有转向，把修改之后的nx、ny赋值，其实最后的代码没变。  


```cpp
class Solution
{
public:
    vector<vector<int>> generateMatrix(int n)
    {
        // 1. 初始化结果矩阵
        vector<vector<int>> res(n, vector<int>(n, 0));

        // 2. 定义方向数组：右、下、左、上（顺时针）
        int dx[4] = {0, 1, 0, -1}; // 行的变化
        int dy[4] = {1, 0, -1, 0}; // 列的变化

        // 3. 初始位置和方向
        int x = 0, y = 0; // 从 (0,0) 开始
        int dir = 0;      // 0:右, 1:下, 2:左, 3:上

        // 4. 填充数字 1 到 n*n
        for (int i = 1; i <= n * n; i++)
        {
            // 填充当前位置
            res[x][y] = i;

            // 计算下一个位置
            int nx = x + dx[dir];
            int ny = y + dy[dir];

            // 5. 判断是否需要转向
            // 条件：越界 或者 下一个位置已经被填充
            // nx < 0往上和左越界，nx >= n往右和往下越界，res[nx][ny]s != 0重复
            while (nx < 0 || nx >= n || ny < 0 || ny >= n || res[nx][ny] != 0)
            {
                // 转向：顺时针旋转90度
                dir = (dir + 1) % 4;
                // 重新计算下一个位置
                nx = x + dx[dir];
                ny = y + dy[dir];
            }

            // 移动到下一个位置
            x = nx;
            y = ny;
        }

        return res;
    }
};
```
