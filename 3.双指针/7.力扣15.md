给一个整数数组 nums ，判断是否存在三元组 [nums[i], nums[j], nums[k]] 满足 i != j、i != k 且 j != k ，同时还满足 nums[i] + nums[j] + nums[k] == 0 。请返回所有和为 0 且不重复的三元组。  
注意：不仅要求索引不一样，还要三个元素的值不一样  

示例：  
输入：nums = [-1,0,1,2,-1,-4]    
输出：[[-1,-1,2],[-1,0,1]]  
解释：  
nums[0] + nums[1] + nums[2] = (-1) + 0 + 1 = 0 。  
nums[1] + nums[2] + nums[4] = 0 + 1 + (-1) = 0 。  
nums[0] + nums[3] + nums[4] = (-1) + 2 + (-1) = 0 。  
不同的三元组是 [-1,0,1] 和 [-1,-1,2] 。  

***三数之和***

deepseek的解法：  
先排序，然后分别对每个元素进行查找是否有另外两个元素能满足三者相加为0，即固定a，在a的循环中b和c对撞寻找。其中细分条件：若三者相加 <0，则让小的大点，左指针右移；若 >0，则让大的小店，右指针左移    
注意对撞指针在移动指针的时候要加条件left < right  

```cpp  
class Solution
{
public:
    vector<vector<int>> threeSum(vector<int> &nums)
    {
        // 放答案的
        vector<vector<int>> results;
        // 若少于三个则无解
        int size = nums.size();
        if (size < 3)
        {
            return {};
        }

        sort(nums.begin(), nums.end());
        for (int i = 0; i < size; i++)
        {
            if (nums[i] > 0) // 因为排序过，往后都不可能有<0的
                break;
            if (i > 0 &&nums[i] == nums[i - 1]) // 和上一个同，跳过
                continue;

            int l = i + 1;
            int r = size - 1;
            while (l < r)
            {
                int sum = nums[i] + nums[l] + nums[r];
                if (sum < 0)
                    l++;
                else if (sum > 0)
                    r--;
                else
                {
                    results.push_back({nums[i], nums[l], nums[r]});

                    while (l < r && nums[l] == nums[l + 1])
                        l++; // 移动到最后一个的该数值位置
                    l++; // 再移动到下一个元素，非该数值，即去重

                    while (l < r && nums[r] == nums[r - 1])
                        r--; // 移动到最前一个的该数值位置
                    r--; // 再移动到上一个元素，非该数值，即去重
                }
            }
        }
        return results;
    }
};
```


官方的解法：  
与上面不同的是，这个是固定a，然后在a的循环中固定b，在b的循环中调整c   

```cpp
class Solution {
public:
    vector<vector<int>> threeSum(vector<int>& nums) {
        int n = nums.size();
        sort(nums.begin(), nums.end());
        vector<vector<int>> ans;
        // 枚举 a
        for (int first = 0; first < n; ++first) {
            // 需要和上一次枚举的数不相同
            if (first > 0 && nums[first] == nums[first - 1]) {
                continue;
            }
            // c 对应的指针初始指向数组的最右端
            int third = n - 1;
            int target = -nums[first];
            // 枚举 b
            for (int second = first + 1; second < n; ++second) {
                // 需要和上一次枚举的数不相同
                if (second > first + 1 && nums[second] == nums[second - 1]) {
                    continue;
                }
                // 需要保证 b 的指针在 c 的指针的左侧
                while (second < third && nums[second] + nums[third] > target) {
                    --third;
                }
                // 如果指针重合，随着 b 后续的增加
                // 就不会有满足 a+b+c=0 并且 b<c 的 c 了，可以退出循环
                if (second == third) {
                    break;
                }
                if (nums[second] + nums[third] == target) {
                    ans.push_back({nums[first], nums[second], nums[third]});
                }
            }
        }
        return ans;
    }
};
```
